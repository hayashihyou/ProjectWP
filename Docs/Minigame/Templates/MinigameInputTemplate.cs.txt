using UnityEngine;
using UnityEngine.InputSystem;

/// <summary>
/// ★1 この入力の説明(例: Aボタンでストップウォッチを止める)
/// ボタンが押されたら、自分の行動をサーバーに送る。Netcode は使わない。シーンのどこに置いてもよい
/// </summary>
public class MinigameInputTemplate : MonoBehaviour // ★2 クラス名を変える(ファイル名と同じにする)
{
    [Tooltip("押したときにサーバーへ送る行動の名前。担当者の OnServerPlayerAction で見る名前と同じにする")]
    [SerializeField] private string actionName = "Stop"; // ★3 行動の名前

    [Tooltip("使うボタン(Assets/Input の Input Actions からドラッグする)")]
    [SerializeField] private InputActionReference button; // ★4 使うボタン

    private void OnEnable()
    {
        if (button == null) { return; }
        button.action.performed += HandlePressed;
        button.action.Enable();
    }

    private void OnDisable()
    {
        if (button == null) { return; }
        button.action.performed -= HandlePressed;
    }

    private void HandlePressed(InputAction.CallbackContext context)
    {
        Send();
    }

    /// <summary>行動をサーバーに送る。画面のボタン(UI の Button の OnClick)からも呼べる</summary>
    public void Send()
    {
        // 今操作してよいとき(プレイ中で、脱落していない)だけ送る
        if (!MinigameInfo.CanControl) { return; }

        MinigameInfo.SendAction(actionName); // ★5 値も送りたいときは SendAction(actionName, 値)
    }
}
